cmake_minimum_required(VERSION 3.20)
project(GalaxyExpansion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(json)
add_library(pool src/object_pool.cpp)
target_include_directories(pool PUBLIC src)
add_executable(galaxy src/main.cpp)
target_link_libraries(galaxy PRIVATE pool nlohmann_json::nlohmann_json)
enable_testing()
add_executable(pool_tests tests/pool_tests.cpp)
target_link_libraries(pool_tests PRIVATE pool)
add_test(NAME pool_tests COMMAND pool_tests)
